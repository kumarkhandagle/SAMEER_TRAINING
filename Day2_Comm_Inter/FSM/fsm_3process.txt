`timescale 1ns / 1ps
//////////////////////////////////////////////////////////////////////////////////



module fsm_part1(
input clk,
input rst,
output reg led
    );
    
parameter IDLE = 0, LED_ON = 1, DELAY1 = 2, LED_OFF = 3, DELAY2 =4;   
reg [2:0] state;

reg [26:0] count;
    
always@(posedge clk)
begin
if(rst == 1'b1)
begin
state <= IDLE;
count <= 0;
end
else
begin
            case(state)
            IDLE: begin
            state <= LED_ON;
            end
            
            LED_ON: begin
            state <= DELAY1;
            led   <= 1'b1;
            end
            
            DELAY1:begin
                if(count < 100_000_000)
                begin
                    count <= count + 1;
                end
                else
                begin
                   count <= 0;
                   state <= LED_OFF;
                end
            end
            
            LED_OFF:begin
            state <= DELAY2;
            led   <= 1'b0;
            end
            
           DELAY2:begin
                if(count < 100_000_000)
                begin
                    count <= count + 1;
                end
                else
                begin
                   count <= 0;
                   state <= IDLE;
                end
            end
            
            default: begin
            state <= IDLE;
            led   <= 1'b0;
            
            end
            
            endcase
    
end


end    
    
    
    
    
endmodule
