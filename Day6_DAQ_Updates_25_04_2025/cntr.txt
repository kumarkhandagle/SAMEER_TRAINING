`timescale 1ns / 1ps


module daq_controller(
input wire         clk, rst_n,
input wire         dac_done, adc_done,
input wire         start,
input [11:0]       dac_in,
input wire [11:0]  adc_in,
output wire        dac_en_o,
output wire [11:0] data_to_dac_o,
output wire        adc_en_o,
output wire [11:0] data_from_adc_o,
output reg         done
);


reg [11:0] data_to_dac;
reg        dac_en;
reg        adc_en;
reg [11:0] data_from_adc;


assign dac_en_o        = dac_en;
assign adc_en_o        = adc_en;
assign data_to_dac_o   = data_to_dac;
assign data_from_adc_o = data_from_adc;


parameter IDLE = 0,
          SAMPLE_DATA = 1,
          ENABLE_DAC  = 2,
          ENABLE_ADC  = 3,
          DONE        = 4;

reg [2:0] state,nstate;

/////reset decoder
always@(posedge clk or negedge rst_n)
begin
if(!rst_n)
state <= IDLE;
else
state <= nstate;
end

////next state decoder

always@(*)
begin
case(state)
IDLE: begin
if(start)
nstate = SAMPLE_DATA;
else
nstate = IDLE;
end

SAMPLE_DATA:begin
nstate = ENABLE_DAC;
end

ENABLE_DAC:begin
if(dac_done)
nstate = ENABLE_ADC;
else
nstate = ENABLE_DAC;
end

ENABLE_ADC:begin
if(adc_done)
nstate = DONE;
else
nstate = ENABLE_ADC;
end

DONE:begin
nstate = IDLE;
end

default : nstate = IDLE;
endcase
end

///output decoder
always@(*) begin
case(state)
IDLE:begin
dac_en        = 1'b0;
data_to_dac   = 12'h0;
adc_en        = 1'b0;
data_from_adc = 12'h0;
done          = 1'b0; 
end

SAMPLE_DATA:begin
data_to_dac   = dac_in;
dac_en        = 1'b0;
adc_en        = 1'b0;
data_from_adc = 12'h0;
done          = 1'b0;
end


ENABLE_DAC:begin
dac_en        = 1'b1;
adc_en        = 1'b0;
data_from_adc = 12'h0;
done          = 1'b0;
end

ENABLE_ADC:begin
dac_en        = 1'b0;
data_to_dac   = 12'h0;
adc_en        = 1'b1;
data_from_adc = 0;
done          = 1'b0;
if(adc_done)
data_from_adc = adc_in;
end

DONE:begin
dac_en = 1'b0;
adc_en = 1'b0;
done   = 1'b1;
end

default:begin
dac_en        = 1'b0;
data_to_dac   = 12'h0;
adc_en        = 1'b0;
data_from_adc = 12'h0;
done          = 1'b0;
end
endcase
end

endmodule

`timescale 1ns / 1ps

module tb_daq_controllern;

// Clock period (100 MHz)
parameter CLK_PERIOD = 10;

// DUT Inputs
reg clk;
reg rst_n;
reg start;
reg dac_done;
reg adc_done;
reg [11:0] dac_in;
reg [11:0] adc_in;

// DUT Outputs
wire dac_en_o;
wire [11:0] data_to_dac_o;
wire adc_en_o;
wire [11:0] data_from_adc_o;
wire done;

// Instantiate the DUT
daq_controller uut (
    .clk(clk),
    .rst_n(rst_n),
    .start(start),
    .dac_done(dac_done),
    .adc_done(adc_done),
    .dac_in(dac_in),
    .adc_in(adc_in),
    .dac_en_o(dac_en_o),
    .data_to_dac_o(data_to_dac_o),
    .adc_en_o(adc_en_o),
    .data_from_adc_o(data_from_adc_o),
    .done(done)
);

// Clock generation
initial begin
    clk = 0;
    forever #(CLK_PERIOD/2) clk = ~clk;
end

// Stimulus
initial begin
    // Initialize inputs
    rst_n = 0;
    start = 0;
    dac_done = 0;
    adc_done = 0;
    dac_in = 12'hABC;
    adc_in = 12'h123;

    // Apply reset
    #20;
    rst_n = 1;

    // Wait a bit then start
    #20;
    start = 1;


    // Wait for ENABLE_DAC and set dac_done after some delay
    wait(dac_en_o == 1);
    #30;
    dac_done = 1;
    #10;
    dac_done = 0;

    // Wait for ENABLE_ADC and set adc_done after some delay
    wait(adc_en_o == 1);
    #30;
    adc_done = 1;
    #5;

    // Wait for DONE state
    wait(done == 1);
    #20;
    $finish;
end

endmodule
