module spi_adc_dac_top(
    input  wire        sys_clk,    // System clock for both ADC and DAC
    input  wire        rst_n,      // Active-low reset
    input  wire        start,      // Start signal for both
    output wire [11:0] adc_data_out, // Output from ADC
    output wire        done_dac,
    output wire        done_adc
);

    // Internal wires
    wire        mosi1;      // DAC MOSI connected to ADC DIN (loopback)
    wire        sclk;       // Shared SCLK
    wire        cs_n_dac;   // Chip Select for DAC
    wire        cs_n_adc;   // Chip Select for ADC
    wire [11:0] dac_data_in; // Data going into DAC
    wire        adc_ready;  // ADC ready signal (not used)

    // Instantiate DAC
    spi_dac u_spi_dac (
        .sys_clk (sys_clk),
        .rst_n   (rst_n),
        .mosi1   (mosi1),
        .sclk    (sclk),
        .cs_n    (cs_n_dac),
        .dac_in1 (dac_data_in),
        .start   (start),
        .done    (done_dac)
    );

    // Instantiate ADC
    spi_adc u_spi_adc (
        .clk     (sys_clk),
        .reset_n (rst_n),
        .sample  (start),
        .din1    (mosi1),
        .cs_n    (cs_n_adc),
        .sclk    (sclk),            // Using shared sclk
        .ready   (adc_ready),
        .done    (done_adc),
        .adc1_dout (adc_data_out)
    );

    // Memory and data generation for DAC
    reg [11:0] mem [0:1023];
    reg [11:0] dataOut;
    reg [9:0]  arrCount;

    always @(posedge sclk or negedge rst_n) begin
        if (!rst_n) begin
            $readmemh("C:/Users/kumar/Desktop/dataSine.mem", mem);
            arrCount <= 10'd0;
            dataOut <= 12'd0;
        end
        else if (done_dac) begin
            dataOut  <= mem[arrCount]; 
            arrCount <= arrCount + 1;
        end
    end

    assign dac_data_in = dataOut;

endmodule

`timescale 1ns/1ps

module tb_spi_adc_dac_top;

    reg sys_clk;
    reg rst_n;
    reg start;
    wire [11:0] adc_data_out;
    wire done_dac;
    wire done_adc;

    // Instantiate the top module
    spi_adc_dac_top uut (
        .sys_clk      (sys_clk),
        .rst_n        (rst_n),
        .start        (start),
        .adc_data_out (adc_data_out),
        .done_dac     (done_dac),
        .done_adc     (done_adc)
    );

    // Generate 50 MHz system clock (period = 20 ns)
    initial begin
        sys_clk = 0;
        forever #10 sys_clk = ~sys_clk;
    end

    // Main stimulus
    initial begin
        // Initialize inputs
        rst_n = 0;
        start = 0;

        // Hold reset low for a while
        #100;
        rst_n = 1;

        // Wait a bit after reset
        #50;

        // Send start pulse
        start = 1;


    end

    // Optional: Monitor important signals
    initial begin
        $monitor("Time=%0t | Start=%b | Done_DAC=%b | Done_ADC=%b | ADC_Data_Out=%h", 
                  $time, start, done_dac, done_adc, adc_data_out);
    end

endmodule
