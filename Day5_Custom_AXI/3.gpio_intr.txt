/*
 * helloworld.c: simple test application
 *
 * This application configures UART 16550 to baud rate 9600.
 * PS7 UART (Zynq) is not initialized by this application, since
 * bootrom/bsp configures it to baud rate 115200
 *
 * ------------------------------------------------
 * | UART TYPE   BAUD RATE                        |
 * ------------------------------------------------
 *   uartns550   9600
 *   uartlite    Configurable only in HW design
 *   ps7_uart    115200 (configured by bootrom/bsp)
 */

#include <stdio.h>
#include "platform.h"
#include "xil_printf.h"
#include "xscugic.h"
#include "xparameters.h"
#include "xgpio.h"

XScuGic gicHandler;
XGpio ledInst;
XGpio swInst;

void sw_handler() {
    // Only process if Channel 1 caused the interrupt
    if ((XGpio_InterruptGetStatus(&swInst) & XGPIO_IR_CH1_MASK) != XGPIO_IR_CH1_MASK) {
        return;
    }

    // Disable and clear interrupt
    XGpio_InterruptDisable(&swInst, XGPIO_IR_CH1_MASK);
    (void) XGpio_InterruptClear(&swInst, XGPIO_IR_CH1_MASK);

    // Read switch value and set to LED
    u32 btn_value = XGpio_DiscreteRead(&swInst, 1);
    xil_printf("Interrupt received. Switch = %d\n\r", btn_value);
    XGpio_DiscreteWrite(&ledInst, 1, btn_value);

    // Re-enable interrupt
    XGpio_InterruptEnable(&swInst, XGPIO_IR_CH1_MASK);
}




void peri_init(){


	s32 status;
	XGpio_Config *ledConfig;
	XGpio_Config *swConfig;

	////// Sw initialization
	swConfig = XGpio_LookupConfig(XPAR_AXI_GPIO_0_DEVICE_ID);
    status = XGpio_CfgInitialize(&swInst, swConfig, swConfig->BaseAddress);
		if (status == XST_SUCCESS)
				xil_printf("SW init Successful\n\r");
	    else
				xil_printf("SW init Failed\n\r");
    XGpio_SetDataDirection(&swInst, 1, 0xFF);

	//// LED initialization
	ledConfig = XGpio_LookupConfig(XPAR_AXI_GPIO_1_DEVICE_ID);
	status = XGpio_CfgInitialize(&ledInst, ledConfig, ledConfig->BaseAddress);
	if(status == XST_SUCCESS)
			xil_printf("LED init Successful\n\r");
		else
			xil_printf("LED init Failed\n\r");
	XGpio_SetDataDirection(&ledInst, 1, 0x00);

	////// notify cpu to ready for exception, register handler for intr , enable specific type of intr
	 Xil_ExceptionInit(); ///Initializes the exception vector table

	  //Registers your interrupt controller’s handler for IRQ exceptions.
     Xil_ExceptionRegisterHandler(XIL_EXCEPTION_ID_IRQ_INT,  //// interrupt request from hardware iRQ
				(Xil_ExceptionHandler) XScuGic_InterruptHandler,
				 &gicHandler);
     ///// unmask irq, enable irq globally
     Xil_ExceptionEnable();

     //// creating instance of GIC
     XScuGic_Config *gicConfig;
     gicConfig  = XScuGic_LookupConfig(XPAR_PS7_SCUGIC_0_DEVICE_ID);
     status = XScuGic_CfgInitialize(&gicHandler, gicConfig, gicConfig->CpuBaseAddress);
     if (status == XST_SUCCESS)
     {
    	 xil_printf("GIC init done\n\r");
     }
     else
     {
    	 xil_printf("GIC init failed\n\r");
     }

     ///////specify handler to be executed when specified peripheral send intr reques

     status = XScuGic_Connect(&gicHandler, XPAR_FABRIC_GPIO_0_VEC_ID,
     			(Xil_ExceptionHandler)sw_handler, (void *) &swInst);

     XScuGic_Enable(&gicHandler, XPAR_FABRIC_GPIO_0_VEC_ID);

 	 XGpio_InterruptEnable(&swInst, 1);
 	 XGpio_InterruptGlobalEnable(&swInst);




}

int main()
{
    init_platform();
    peri_init();

    print("UART init done\n\r");

    while(1);

    cleanup_platform();
    return 0;
}
